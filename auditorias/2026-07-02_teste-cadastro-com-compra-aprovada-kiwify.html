3 — PRE_SCHEMA_CHECK

ℹ️
Executar no Supabase SQL Editor antes de qualquer inserção de teste. Confirmar colunas, tipos e valores reais.
-- Estrutura das tabelas envolvidas no fluxo de compra
SELECT
  table_name,
  column_name,
  data_type,
  is_nullable
FROM information_schema.columns
WHERE table_schema = 'public'
  AND table_name IN (
    'kiwify_purchases',
    'kiwify_products',
    'users',
    'tenants',
    'subscriptions',
    'credits_wallet',
    'credits_ledger'
  )
ORDER BY table_name, ordinal_position;

-- Constraint UNIQUE de provider_order_id
SELECT
  tc.table_name, tc.constraint_name, tc.constraint_type, kcu.column_name
FROM information_schema.table_constraints tc
JOIN information_schema.key_column_usage kcu
  ON tc.constraint_name = kcu.constraint_name
  AND tc.table_schema   = kcu.table_schema
WHERE tc.table_schema = 'public'
  AND tc.table_name   = 'kiwify_purchases'
  AND tc.constraint_type IN ('UNIQUE', 'PRIMARY KEY');

-- Valores reais de status na tabela
SELECT DISTINCT status FROM public.kiwify_purchases ORDER BY status;

-- Produtos ativos e seus product_keys
SELECT
  kiwify_product_id, product_name, product_key,
  plan_code, billing_cycle, is_active
FROM public.kiwify_products
WHERE is_active = true
ORDER BY display_order;

4 — Localizar Compra de Teste Existente (somente leitura)

⚠️
Execute somente leitura. Não altere dados. Não exiba e-mail de cliente sem autorização.

Resumo agregado de compras disponíveis (sem expor e-mails)

SELECT
  count(*)      AS total_aprovadas_sem_ativar,
  product_key,
  plan_code,
  billing_cycle,
  min(paid_at)  AS mais_antiga,
  max(paid_at)  AS mais_recente
FROM public.kiwify_purchases
WHERE status       = 'APPROVED'
  AND activated_at IS NULL
  AND product_key != 'UNKNOWN'
GROUP BY product_key, plan_code, billing_cycle
ORDER BY max(paid_at) DESC;

7 — Queries de Verificação (somente leitura)

ℹ️
Substituir EMAIL_DE_TESTE pelo e-mail usado no teste. Executar no SQL Editor após o cadastro.

auth.users — conta criada

SELECT
  id, email, created_at,
  email_confirmed_at,
  last_sign_in_at
FROM auth.users
WHERE email = lower(trim('EMAIL_DE_TESTE'));

public.users — perfil e tenant

SELECT
  id, email, name, tenant_id, plan, created_at
FROM public.users
WHERE email = lower(trim('EMAIL_DE_TESTE'));

tenants — tenant criado

SELECT
  t.id, t.name, t.created_at
FROM public.tenants t
JOIN public.users u ON u.tenant_id = t.id
WHERE u.email = lower(trim('EMAIL_DE_TESTE'));

subscriptions — plano ativo

SELECT
  s.id, s.status, s.billing_cycle,
  s.current_period_end, s.provider,
  p.name AS plan_name,
  s.updated_at
FROM public.subscriptions s
JOIN public.plans p ON p.id = s.plan_id
JOIN public.users u ON u.tenant_id = s.tenant_id
WHERE u.email = lower(trim('EMAIL_DE_TESTE'));

kiwify_purchases — compra ativada

SELECT
  lower(email)                        AS email,
  status,
  product_key,
  plan_code,
  paid_at,
  activated_at,
  tenant_id,
  left(provider_order_id, 10) || '...' AS order_parcial
FROM public.kiwify_purchases
WHERE lower(trim(email)) = lower(trim('EMAIL_DE_TESTE'))
ORDER BY paid_at DESC NULLS LAST;

credits_wallet — saldo

SELECT
  cw.balance, cw.updated_at
FROM public.credits_wallet cw
JOIN public.users u ON u.tenant_id = cw.tenant_id
WHERE u.email = lower(trim('EMAIL_DE_TESTE'));

credits_ledger — histórico de movimentações

SELECT
  cl.amount, cl.type, cl.description, cl.source, cl.created_at
FROM public.credits_ledger cl
JOIN public.users u ON u.tenant_id = cl.tenant_id
WHERE u.email = lower(trim('EMAIL_DE_TESTE'))
ORDER BY cl.created_at DESC;
